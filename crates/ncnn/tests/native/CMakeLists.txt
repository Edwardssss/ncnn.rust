cmake_minimum_required(VERSION 3.15)
project(ncnn_cpu_reference LANGUAGES CXX)

if("$ENV{NCNN_INSTALL_DIR}" STREQUAL "")
    message(FATAL_ERROR "Set NCNN_INSTALL_DIR to the installation used by the Rust build")
endif()
set(ncnn_DIR "$ENV{NCNN_INSTALL_DIR}/lib/cmake/ncnn")
if(NOT EXISTS "${ncnn_DIR}/ncnnConfig.cmake")
    set(ncnn_DIR "$ENV{NCNN_INSTALL_DIR}/lib64/cmake/ncnn")
endif()
find_package(ncnn CONFIG REQUIRED NO_DEFAULT_PATH)
add_executable(ncnn_cpu_reference reference.cpp)
target_compile_features(ncnn_cpu_reference PRIVATE cxx_std_17)
target_compile_definitions(ncnn_cpu_reference PRIVATE NOMINMAX)
target_link_libraries(ncnn_cpu_reference PRIVATE ncnn)
